'use strict';
const $=id=>document.getElementById(id), enc=new TextEncoder(), dec=new TextDecoder(), C=crypto.subtle;
const b64=bytes=>btoa(String.fromCharCode(...new Uint8Array(bytes)));
const un64=str=>Uint8Array.from(atob(str),c=>c.charCodeAt(0));
const utf=s=>enc.encode(s), err=e=>{$('status').textContent='Error: '+(e?.message||e)};
let mine=null;
async function digest(s){return [...new Uint8Array(await C.digest('SHA-256',utf(s)))].map(x=>x.toString(16).padStart(2,'0')).join('').match(/.{1,4}/g).join(' ')}
function canonical(x){if(Array.isArray(x))return '['+x.map(canonical).join(',')+']';if(x&&typeof x==='object')return '{'+Object.keys(x).sort().map(k=>JSON.stringify(k)+':'+canonical(x[k])).join(',')+'}';return JSON.stringify(x)}
async function importCard(raw){let v=JSON.parse(raw);if(v.type!=='PLCONTACT1'||!v.enc||!v.sig||v.enc.kty!=='RSA'||v.sig.kty!=='EC')throw Error('Invalid public contact card');let encrypt=await C.importKey('jwk',v.enc,{name:'RSA-OAEP',hash:'SHA-256'},false,['encrypt']);let verify=await C.importKey('jwk',v.sig,{name:'ECDSA',namedCurve:'P-256'},false,['verify']);return {v,encrypt,verify,fp:await digest(canonical(v))}}
async function publicCard(encKey,sigKey){return {type:'PLCONTACT1',enc:await C.exportKey('jwk',encKey),sig:await C.exportKey('jwk',sigKey)}}
function showIdentity(){ $('public').value=JSON.stringify(mine.card);$('identity').textContent='Identity loaded. Fingerprint: '+mine.fingerprint; }
$('generate').onclick=async()=>{try{if($('pass').value.length<12)throw Error('Passphrase must be at least 12 characters');$('status').textContent='Generating keys locally…';let a=await C.generateKey({name:'RSA-OAEP',modulusLength:3072,publicExponent:new Uint8Array([1,0,1]),hash:'SHA-256'},true,['encrypt','decrypt']);let b=await C.generateKey({name:'ECDSA',namedCurve:'P-256'},true,['sign','verify']);let card=await publicCard(a.publicKey,b.publicKey);mine={dec:a.privateKey,sign:b.privateKey,card,fingerprint:await digest(canonical(card))};$('save').disabled=false;showIdentity();$('status').textContent='Identity generated. Download encrypted vault NOW.'}catch(e){err(e)}};
async function derive(pass,salt){let base=await C.importKey('raw',utf(pass),'PBKDF2',false,['deriveKey']);return C.deriveKey({name:'PBKDF2',hash:'SHA-256',salt,iterations:600000},base,{name:'AES-GCM',length:256},false,['encrypt','decrypt'])}
$('save').onclick=async()=>{try{if(!mine||$('pass').value.length<12)throw Error('Load keys and supply vault passphrase');let salt=crypto.getRandomValues(new Uint8Array(16)),nonce=crypto.getRandomValues(new Uint8Array(12));let k=await derive($('pass').value,salt);let secrets={dec:await C.exportKey('jwk',mine.dec),sign:await C.exportKey('jwk',mine.sign)};let ciphertext=await C.encrypt({name:'AES-GCM',iv:nonce},k,utf(JSON.stringify(secrets)));let data=JSON.stringify({type:'PLVAULT1',card:mine.card,salt:b64(salt),nonce:b64(nonce),cipher:b64(ciphertext)});let link=document.createElement('a');link.href=URL.createObjectURL(new Blob([data],{type:'application/json'}));link.download='private-line-encrypted-vault.json';link.click();setTimeout(()=>URL.revokeObjectURL(link.href),5000);$('status').textContent='Encrypted vault downloaded. $('check').onclick=async()=>{try{let other=await importCard($('recipient').value);$('fingerprint').textContent='Contact fingerprint: '+other.fp+' — verify separately';$('status').textContent='Valid public card.'}catch(e){err(e)}};
$('encrypt').onclick=async()=>{try{if(!mine)throw Error('First create/unlock your identity');let other=await importCard($('recipient').value);let aes=await C.generateKey({name:'AES-GCM',length:256},true,['encrypt','decrypt']);let iv=crypto.getRandomValues(new Uint8Array(12));let ct=await C.encrypt({name:'AES-GCM',iv},aes,utf($('plain').value));let raw=await C.exportKey('raw',aes);let wrapped=await C.encrypt({name:'RSA-OAEP'},other.encrypt,raw);let data={type:'PLMESSAGE1',from:mine.fingerprint,to:other.fp,iv:b64(iv),wrapped:b64(wrapped),cipher:b64(ct)};let sig=await C.sign({name:'ECDSA',hash:'SHA-256'},mine.sign,utf(canonical(data)));$('out').value=JSON.stringify({...data,signature:b64(sig)});$('status').textContent='Message encrypted and signed locally.'}catch(e){err(e)}};
